## 구현 계획

### 수정 파일 목록

| # | 파일 | 변경 종류 |

| 1 | `tdc_drv_iqs323.h` | 후보 구조체·extern 선언, API 선언 추가 |

### Step 1 — `tdc_drv_iqs323.h` 추가 선언

`TDC_BOARD_VARIANT` 상수 블록 이후에 다음 추가:

```c
#if defined(TDC_TOUCH_ATI_CALIB_MODE)

#define TDC_DRV_IQS323_CALIB_CANDIDATE_COUNT  10

typedef struct {
    uint8_t mult_lsb;
    uint8_t mult_msb;
    uint8_t comp_lsb;
    uint8_t comp_msb;
} tdc_drv_iqs323_calib_candidate_t;

extern const tdc_drv_iqs323_calib_candidate_t
    tdc_drv_iqs323_calib_candidates[TDC_DRV_IQS323_CALIB_CANDIDATE_COUNT];

uint8_t tdc_drv_iqs323_calib_find_candidate(void);

#endif /* TDC_TOUCH_ATI_CALIB_MODE */
```

### Step 2 — `tdc_drv_iqs323.c` 후보 테이블 + 함수 추가

파일 말미(`TDC_BOARD_VARIANT` 매크로 블록 이후)에 추가:

```c
/* ────────────────────────────────────────────────────────────────
 *  ATI Calibration Candidate Table
 *  (TDC_TOUCH_ATI_CALIB_MODE 빌드 전용)
 * ──────────────────────────────────────────────────────────────── */
#if defined(TDC_TOUCH_ATI_CALIB_MODE)

const tdc_drv_iqs323_calib_candidate_t
    tdc_drv_iqs323_calib_candidates[TDC_DRV_IQS323_CALIB_CANDIDATE_COUNT] =
{
    /* mult_lsb  mult_msb  comp_lsb  comp_msb  비고              */
    { 0x82,     0x5C,     0x00,     0x48 },  /* 1: MINI 이하      */
    { 0x82,     0x5E,     0xFF,     0x53 },  /* 2: ≈DEVELOP       */
    { 0x82,     0x5A,     0x00,     0x58 },  /* 3: ≈MINI          */
    { 0x82,     0x5C,     0x00,     0x60 },  /* 4                 */
    { 0x82,     0x5C,     0x00,     0x68 },  /* 5                 */
    { 0x82,     0x5C,     0x00,     0x70 },  /* 6                 */
    { 0x82,     0x5E,     0xFF,     0x73 },  /* 7: ≈PACKAGE       */
    { 0x82,     0x5A,     0x00,     0x80 },  /* 8                 */
    { 0x82,     0x58,     0x00,     0x90 },  /* 9                 */
    { 0x82,     0x56,     0x00,     0xA0 },  /* 10: 고용량 조립    */
};

uint8_t tdc_drv_iqs323_calib_find_candidate(void)
{
    uint8_t mult_lsb, mult_msb, comp_lsb, comp_msb;

    if (!read_register(TDC_DRV_IQS323_REG_ADDR_SENSOR0_ATI_MULT, &mult_lsb, &mult_msb))
    {
        ci_printe("[TOUCH] CALIB: MULT READ FAIL \r\n");
        return 0;
    }
    if (!read_register(TDC_DRV_IQS323_REG_ADDR_SENSOR0_ATI_COMP, &comp_lsb, &comp_msb))
    {
        ci_printe("[TOUCH] CALIB: COMP READ FAIL \r\n");
        return 0;
    }

    ci_printi("[TOUCH] CALIB READ — MULT: LSB=0x%02X MSB=0x%02X  COMP: LSB=0x%02X MSB=0x%02X \r\n",
              mult_lsb, mult_msb, comp_lsb, comp_msb);

    uint32_t comp16_actual = ((uint32_t)comp_msb << 8) | comp_lsb;
    uint8_t  best_idx      = 1;
    uint32_t best_dist     = UINT32_MAX;

    for (uint8_t i = 0; i < TDC_DRV_IQS323_CALIB_CANDIDATE_COUNT; i++)
    {
        const tdc_drv_iqs323_calib_candidate_t *c = &tdc_drv_iqs323_calib_candidates[i];
        uint32_t comp16_cand = ((uint32_t)c->comp_msb << 8) | c->comp_lsb;

        int32_t  dc   = (int32_t)comp16_actual - (int32_t)comp16_cand;
        int32_t  dm   = (int32_t)mult_msb      - (int32_t)c->mult_msb;
        uint32_t dist = (uint32_t)(dc * dc) + (uint32_t)(dm * dm * 256);

        if (dist < best_dist)
        {
            best_dist = dist;
            best_idx  = i + 1;
        }
    }

    ci_printi("[TOUCH] CALIB: Best candidate = %d \r\n", best_idx);
    return best_idx;
}

#endif /* TDC_TOUCH_ATI_CALIB_MODE */
```

### Step 3 — `tdc_touch.c` 수정

#### 3-a. 헤더 include 추가 (파일 상단)

```c
#if defined(TDC_TOUCH_ATI_CALIB_MODE)
#include "LedOutput.h"
#endif
```

#### 3-b. LED 블링크 helper 함수 추가 (static, 파일 내부)

```c
#if defined(TDC_TOUCH_ATI_CALIB_MODE)

#define TDC_TOUCH_CALIB_BLINK_ON_MS   400
#define TDC_TOUCH_CALIB_BLINK_OFF_MS  200
#define TDC_TOUCH_CALIB_PAUSE_MS      800

static void calib_delay_ms(int ms)
{
    int t = ci_timer_get_tick();
    while (ms > (ci_timer_get_tick() - t))
    {
        SYS_WATCHDOG_REFRESH();
    }
}

static void tdc_touch_calib_led_report(uint8_t idx)
{
    for (uint8_t i = 0; i < idx; i++)
    {
        turnON_GreenLED();
        calib_delay_ms(TDC_TOUCH_CALIB_BLINK_ON_MS);
        LED_black();
        calib_delay_ms(TDC_TOUCH_CALIB_BLINK_OFF_MS);
    }
    calib_delay_ms(TDC_TOUCH_CALIB_PAUSE_MS);
    turnON_RedLED();
}

#endif /* TDC_TOUCH_ATI_CALIB_MODE */
```

#### 3-c. `try_finish_init()` calib 분기 추가

ATI 완료 확인 후, `tdc_drv_iqs323_apply_settings()` 호출 직전:

```c
    SYS_WATCHDOG_REFRESH();

#if defined(TDC_TOUCH_ATI_CALIB_MODE)
    /* [CALIB MODE] auto-ATI 직후 IC 자체 결정값 읽기 → LED 결과 출력 → 무한 루프 */
    {
        uint8_t idx = tdc_drv_iqs323_calib_find_candidate();
        if (idx == 0) { idx = 1; }  /* read fail 시 fallback */
        tdc_touch_calib_led_report(idx);
        while (1) { SYS_WATCHDOG_REFRESH(); }
    }
#else
    tdc_drv_iqs323_apply_settings();
#endif

    SYS_WATCHDOG_REFRESH();
```

## 검증 방법

1. `TDC_TOUCH_ATI_CALIB_MODE` 미정의 빌드 → 기존 동작과 동일한지 확인 (터치 정상 동작)
2. `TDC_TOUCH_ATI_CALIB_MODE` 정의 빌드 → 보드에 플래시 후 전원 ON:
   - 녹색 LED가 N회 점멸하는지 확인 (N = 1~10 범위)
   - 이후 빨간 LED 점등 유지 확인
   - N 값을 분석.md 후보 테이블과 대조하여 실제 조립 조건 파악
3. 시리얼 연결 가능한 환경이라면 `ci_printi` 출력도 함께 확인:
   - `CALIB READ — MULT: LSB=... MSB=...  COMP: LSB=... MSB=...`
   - `CALIB: Best candidate = N`

## 주의사항

- `tdc_drv_iqs323_calib_find_candidate()` 은 `static read_register()`를 직접 호출 → 같은 `.c` 파일에 위치해야 함 (Step 2)
- LED arbiter가 활성화된 상태에서 `turnON_GreenLED()` / `LED_black()` 직접 호출은 짧은 점멸 구간에서 arbiter와 충돌 가능성 있음. 단, calib 모드는 `while(1)` 직전 단계이므로 arbiter가 다른 상태를 요청할 일이 없어 실질적 문제 없음.
